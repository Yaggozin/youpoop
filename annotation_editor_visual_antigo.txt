    function timeToSeconds(min, sec) { return (parseFloat(min || 0) * 60) + parseFloat(sec || 0); }
    function secondsToTime(totalSec) {
        totalSec = parseFloat(totalSec);

        const sec = (totalSec % 60).toFixed(2);
        return { min, sec };

    /**
     * Converte HEX para RGBA com transparência fixa de 50% (0.5).
     */
    function hexToRgba(hex, alpha = 0.5) {
        const h = hex.replace('#', '');
        const r = parseInt(h.substring(0, 2), 16);
        const g = parseInt(h.substring(2, 4), 16);
        const b = parseInt(h.substring(4, 6), 16);
        return `rgba(${r}, ${g}, ${b}, ${alpha})`;
    }

    function selectColor(swatch) {
        if (!selectedElement) return;

        document.querySelectorAll('.color-swatch').forEach(el => el.classList.remove('active'));
        swatch.classList.add('active');

        const colorHex = swatch.getAttribute('data-color');
        
        document.getElementById('annoColor').value = colorHex;
        
        applyColorToElement(selectedElement, colorHex);
    }

    // Adicionado: Função para atualizar a cor baseada no tipo (title/spotlight/note)
    function applyColorToElement(el, colorHex) {
        const type = el.dataset.type || document.getElementById('annoType').value; 
        
        if (type !== 'title' && type !== 'spotlight') {
            const colorRgba = hexToRgba(colorHex, 0.5);
            el.style.backgroundColor = colorRgba;
            
            // Lógica para cor de texto contrastante
            if (colorHex.toLowerCase() === '#f1c40f' || colorHex.toLowerCase() === '#ffffff') {
                el.style.color = 'black';
            } else {
                el.style.color = 'white';
            }
        } else {
             el.style.backgroundColor = 'transparent';
             if (type === 'title') {
                 el.style.color = '#fff';
             } else if (type === 'spotlight') {
                 el.style.color = 'transparent'; // O texto some no destaque
                 el.style.borderColor = colorHex; // A cor define a borda
             }
        }
        el.dataset.color = colorHex;
    }

                bar.classList.add('selected-bar');
            }
            
            bar.style.left = `${startPercent}%`;
            bar.style.width = `${durationPercent}%`;
            
            // Atribui o texto e dataset para mapeamento
            bar.innerText = textContent.length > 20 ? textContent.substring(0, 17) + '...' : textContent;
            bar.dataset.elementId = el.id; // Usa o ID (temporário ou DB ID)

            // Evento de clique para selecionar a anotação
            bar.addEventListener('click', (e) => {
                e.stopPropagation(); 
                selectAnnotation(el); 
            });
            
            timelineAnnotationBarsContainer.appendChild(bar);
        });
    }

    function updateFormPositionAndSize(el) {
        const containerRect = wrapper.getBoundingClientRect();
        const elRect = el.getBoundingClientRect();

        const x = ((elRect.left - containerRect.left) / containerRect.width) * 100;
        const y = ((elRect.top - containerRect.top) / containerRect.height) * 100;
        const width = (elRect.width / containerRect.width) * 100;
        const height = (elRect.height / containerRect.height) * 100;

        document.getElementById(INPUT_IDS.X).value = x.toFixed(2);
        document.getElementById(INPUT_IDS.Y).value = y.toFixed(2);
        document.getElementById(INPUT_IDS.WIDTH).value = width.toFixed(2);
        document.getElementById(INPUT_IDS.HEIGHT).value = height.toFixed(2);document.getElementById(INPUT_IDS.X).value = x;
    }

    function updateVisualsFromInput(e) {
        if (!selectedElement) return; 
        const getValue = (id) => parseFloat(document.getElementById(id).value) || 0;
        const targetId = e.target.id;

        if (targetId === INPUT_IDS.WIDTH || targetId === INPUT_IDS.HEIGHT) {
            const w = getValue(INPUT_IDS.WIDTH);
            const h = getValue(INPUT_IDS.HEIGHT);
            if (w > 0) selectedElement.style.width = w + '%';
            if (h > 0) selectedElement.style.height = h + '%';
        } 
        if (targetId === INPUT_IDS.X || targetId === INPUT_IDS.Y) {
            const x = getValue(INPUT_IDS.X);
            const y = getValue(INPUT_IDS.Y);
            selectedElement.style.left = x + '%';
            selectedElement.style.top = y + '%';
        }
    }
    
    function updateFormSizeOnNativeResize(mutationsList, observer) {
        if (selectedElement) {
            const styleChanged = mutationsList.some(mutation => 
                mutation.target === selectedElement && mutation.attributeName === 'style'
            );
            if (styleChanged && !isDragging) {
                setTimeout(() => updateFormPositionAndSize(selectedElement), 0);
            }
        }
    }

    function createAnnotationElement(data) {
        const el = document.createElement('div');
        // Garante que o elemento tenha um ID (temporário ou do DB)
        el.id = data.id ? `anno-${data.id}` : `temp-anno-${Date.now()}-${Math.floor(Math.random() * 1000)}`; 
        
        el.className = `editable-annotation annotation-${data.type}`;
        el.innerText = data.text_content;
        
        el.style.left = data.x_pos + '%';
        el.style.top = data.y_pos + '%';
        el.style.width = data.width + '%';
        el.style.height = data.height + '%';
        
        el.style.fontSize = data.font_size || '14px'; 

        el.dataset.id = data.id || '';
        el.dataset.start = data.start_time_sec;
        el.dataset.end = data.end_time_sec;
        el.dataset.link = data.link_url || '';
        el.dataset.type = data.type;
        el.dataset.color = data.color || '#000000'; 
        el.dataset.fontSize = data.font_size || '14px';

        if (!data.id) selectAnnotation(el); 
        
        // NOVO: Renderiza a timeline para incluir a nova anotação

    }

    async function saveData() {
        if (!selectedElement) return;

        const x_pos = parseFloat(document.getElementById(INPUT_IDS.X).value);
        const y_pos = parseFloat(document.getElementById(INPUT_IDS.Y).value);
        const width = parseFloat(document.getElementById(INPUT_IDS.WIDTH).value);
        const height = parseFloat(document.getElementById(INPUT_IDS.HEIGHT).value);

        const startTime = timeToSeconds(document.getElementById('startMin').value, document.getElementById('startSec').value);
        const endTime = timeToSeconds(document.getElementById('endMin').value, document.getElementById('endSec').value);

        const fontSizeValue = document.getElementById('annoFontSize').value;
        const finalFontSize = parseFloat(fontSizeValue) > 0 ? parseFloat(fontSizeValue) + 'px' : '14px';

        selectedElement.dataset.start = startTime;
        selectedElement.dataset.end = endTime;
        selectedElement.dataset.link = document.getElementById('annoLink').value;
        selectedElement.dataset.color = document.getElementById('annoColor').value;
        selectedElement.dataset.fontSize = finalFontSize;

        const payload = {
            action: 'save',
            id: document.getElementById('annoId').value,
            video_id: videoId,
            type: document.getElementById('annoType').value,
            text_content: document.getElementById('annoText').value,
            start_time_sec: startTime,
            end_time_sec: endTime,
            x_pos: x_pos.toFixed(2),
            y_pos: y_pos.toFixed(2),
            width: width.toFixed(2),
            height: height.toFixed(2),
            link_url: document.getElementById('annoLink').value,
            color: document.getElementById('annoColor').value,
            font_size: finalFontSize
        };

        const btn = document.querySelector('.btn-save');
        btn.innerText = "Salvando...";

        try {
            const response = await fetch('annotation_editor.php', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const result = await response.json();

            if (result.status === 'success') {
                alert('Salvo com sucesso!');
                if (result.id) {
                    // Atualiza o ID do elemento e dataset após salvar no DB
                    selectedElement.dataset.id = result.id;
                    selectedElement.id = `anno-${result.id}`;
                    document.getElementById('annoId').value = result.id;
                }
                renderTimelineAnnotations(); // NOVO: Re-renderiza a timeline
            } else {
                alert('Erro ao salvar: ' + result.message);
            }
        } catch (err) {
            console.error(err);
            alert('Erro de conexão.');
        } finally {
            btn.innerText = "Salvar alterações"; // Ajustado para o texto correto
        }
    }

    async function deleteData() {
        if (!selectedElement) return;
        const id = document.getElementById('annoId').value;

        if (!id) {
            selectedElement.remove();
            resetForm();
            renderTimelineAnnotations(); // NOVO: Re-renderiza a timeline
            return;
        }

        if(!confirm("Tem certeza que deseja apagar?")) return;

        try {
            const response = await fetch('annotation_editor.php', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ action: 'delete', id: id })
            });
            const result = await response.json();

            if (result.status === 'success') {
                selectedElement.remove();
                resetForm();
                renderTimelineAnnotations(); // NOVO: Re-renderiza a timeline
            } else {
                alert('Erro ao deletar: ' + result.message);
            }
        } catch (err) {
            alert('Erro de conexão.');
        }
    }

    wrapper.addEventListener('mousedown', (e) => {
        if (e.target.classList.contains('editable-annotation')) {
            selectAnnotation(e.target); 
            
            if (e.target.style.cursor !== 'se-resize' && e.target.style.cursor !== 'sw-resize' && e.target.style.cursor !== 's-resize') {
                isDragging = true;
                e.preventDefault(); 
            } else {
                isDragging = false; 
                return;
            }
            
            dragStartX = e.clientX;
            dragStartY = e.clientY;
            initialLeftPercent = parseFloat(selectedElement.style.left);
            initialTopPercent = parseFloat(selectedElement.style.top);
        }
    });

    window.addEventListener('mousemove', (e) => {
        if (isDragging && selectedElement) {
            e.preventDefault();
            const parentRect = wrapper.getBoundingClientRect();
            
            const deltaX = e.clientX - dragStartX;
            const deltaY = e.clientY - dragStartY;
            const deltaXPercent = (deltaX / parentRect.width) * 100;
            const deltaYPercent = (deltaY / parentRect.height) * 100;
            
            let newLeft = initialLeftPercent + deltaXPercent;
            let newTop = initialTopPercent + deltaYPercent;

            const currentWidth = parseFloat(selectedElement.style.width);
            const currentHeight = parseFloat(selectedElement.style.height);
            
            newLeft = Math.max(0, Math.min(newLeft, 100 - currentWidth));
            newTop = Math.max(0, Math.min(newTop, 100 - currentHeight));
            
            selectedElement.style.left = newLeft + '%';
            selectedElement.style.top = newTop + '%';

            document.getElementById(INPUT_IDS.X).value = newLeft.toFixed(2); // <--- O PROBLEMA
            document.getElementById(INPUT_IDS.Y).value = newTop.toFixed(2);  // <--- O PROBLEMA

        }
    });

    window.addEventListener('mouseup', () => {
        isDragging = false;
        if (selectedElement) {
            updateFormPositionAndSize(selectedElement);
        }
    });
